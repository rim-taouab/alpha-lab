# Dépendance optionnelle : orchestration du pipeline avec Apache Airflow.
# Séparée de requirements.txt pour garder l'installation de base légère.
#
# Airflow impose d'installer AVEC son fichier de contraintes (sinon les versions
# de ses très nombreuses dépendances entrent en conflit). Installer ainsi :
#
#   AIRFLOW_VERSION=3.1.0
#   PYTHON_VERSION=3.13
#   CONSTRAINT="https://raw.githubusercontent.com/apache/airflow/constraints-${AIRFLOW_VERSION}/constraints-${PYTHON_VERSION}.txt"
#   pip install "apache-airflow[standard]==${AIRFLOW_VERSION}" --constraint "${CONSTRAINT}"
#
# Airflow >= 3.1 est requis pour Python 3.13.
apache-airflow[standard]==3.1.0
